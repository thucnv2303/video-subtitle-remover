#!/bin/sh
set -eu

staged="$(git diff --cached --name-only --diff-filter=ACMR)"

code_changed="$(
    printf '%s\n' "$staged" |
    grep -E '^(api/|src/|package\.json$|package-lock\.json$)' ||
    true
)"

# Không có source code được stage thì không cần kiểm memory.
if [ -z "$code_changed" ]; then
    exit 0
fi

missing=""

for file in \
    ".ai/current_state.md" \
    ".ai/task_current.md" \
    ".ai/handoff.md"
do
    if ! printf '%s\n' "$staged" | grep -Fxq "$file"; then
        missing="${missing}
- ${file}"
    fi
done

if [ -n "$missing" ]; then
    printf '\nAGENTOS GATE: COMMIT BI CHAN.\n'
    printf 'Source code da duoc stage nhung cac file memory sau chua duoc cap nhat:%s\n' "$missing"
    printf '\nAnti phai cap nhat ket qua vao .ai truoc khi commit.\n'
    printf 'Sau do chay:\n'
    printf 'git add .ai/current_state.md .ai/task_current.md .ai/handoff.md\n\n'
    exit 1
fi

exit 0
